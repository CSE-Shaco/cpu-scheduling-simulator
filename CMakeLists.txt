cmake_minimum_required(VERSION 3.16)
project(cpu_scheduling_simulator LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(scheduling src/simulator.c)
target_include_directories(scheduling PUBLIC include)
target_compile_options(scheduling PRIVATE -Wall -Wextra -Wpedantic)
add_executable(scheduler_c src/main.c)
target_link_libraries(scheduler_c PRIVATE scheduling)
enable_testing()
add_executable(scheduler_c_tests tests/test_simulator.c)
target_link_libraries(scheduler_c_tests PRIVATE scheduling)
add_test(NAME scheduler_correctness COMMAND scheduler_c_tests)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
  add_test(NAME scheduler_cli
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:scheduler_c>)
  add_test(NAME scheduler_tick_oracle
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/differential.py $<TARGET_FILE:scheduler_c>)
  add_test(NAME scheduler_experiments
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_experiments.py $<TARGET_FILE:scheduler_c>)
  set_tests_properties(scheduler_cli scheduler_tick_oracle scheduler_experiments
    PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
